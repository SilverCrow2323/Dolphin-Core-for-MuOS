# ============================================================
# [Core] — Emulation core
# ============================================================
[Core]

# CPUCore : CPU emulation engine.
# CPUCore Parameters : 0 (JIT64), 1 (JITARM64), 2 (Interpreter), 3 (Cached Interpreter), 4 (JITIL, deprecated).
# CPUCore Recommend : 1
# CPUCore Why : Native ARM64 JIT. Mandatory on H700 — the interpreter is 10-50x slower.
CPUCore = 1

# CPUThread : Run CPU and GPU in separate threads (Dual Core).
# CPUThread Parameters : True, False.
# CPUThread Recommend : True
# CPUThread Why : Uses all 4 A53 cores. Non-negotiable.
CPUThread = True

# DSPHLE : Use High-Level Emulation for the DSP.
# DSPHLE Parameters : True, False.
# DSPHLE Recommend : True
# DSPHLE Why : LLE (cycle-accurate) is far too CPU-heavy on A53.
DSPHLE = True

# EnableIdleSkipping : Skip CPU busy-wait loops.
# EnableIdleSkipping Parameters : True, False.
# EnableIdleSkipping Recommend : True
# EnableIdleSkipping Why : Clean win; a handful of games misbehave and need it off.
EnableIdleSkipping = True

# Overclock : Emulated CPU clock multiplier.
# Overclock Parameters : Float (0.5-1.5).
# Overclock Recommend : 1.0 in the baseline; 0.70 in a Performance profile.
# Overclock Why : Lower values reduce CPU load at the cost of game logic speed. Never put an underclock in the baseline.
Overclock = 1.0

# OverclockEnable : Enable the Overclock multiplier.
# OverclockEnable Parameters : True, False.
# OverclockEnable Recommend : False in the baseline.
OverclockEnable = False

# MMU : Emulate the Memory Management Unit.
# MMU Parameters : True, False.
# MMU Recommend : False globally; True for Skies of Arcadia Legends (GEAE8P).
# MMU Why : Expensive. Enable only per-game when a title requires it.
MMU = False

# SyncGPU : Synchronize CPU and GPU threads.
# SyncGPU Parameters : True, False.
# SyncGPU Recommend : False
SyncGPU = False

# SyncOnSkipIdle : Sync after skipping idle.
# SyncOnSkipIdle Parameters : True, False.
SyncOnSkipIdle = True

# Framelimit : Frame rate limiter.
# Framelimit Parameters : 0 (Off), 1 (Auto), 2 (60), 3 (30).
# Framelimit Recommend : 1
Framelimit = 1

# FPRF : Floating-Point Result Flag accuracy.
# FPRF Parameters : True, False.
# FPRF Recommend : False
FPRF = False

# Fastmem : Fast host memory access (virtual-memory mapping).
# Fastmem Parameters : True, False.
# Fastmem Recommend : True
# Fastmem Why : Near-free performance win; safe.
Fastmem = True

# TimingVariance : CPU timing tolerance (higher = more forgiving).
# TimingVariance Parameters : Integer (0-100).
TimingVariance = 40

# AccurateNaNs : Accurate NaN handling (compatibility).
# AccurateNaNs Parameters : True, False.
# AccurateNaNs Recommend : False
AccurateNaNs = False

# EmulationSpeed : Global speed multiplier (1.0 = 100%).
# EmulationSpeed Parameters : Float (0.0-10.0).
EmulationSpeed = 1.0

# SkipIPL : Skip the GameCube IPL boot intro.
# SkipIPL Parameters : True, False.
# SkipIPL Recommend : True
