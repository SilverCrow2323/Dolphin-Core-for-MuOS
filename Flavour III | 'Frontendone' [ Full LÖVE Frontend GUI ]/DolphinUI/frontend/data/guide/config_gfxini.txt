# ============================================================
# GFX.INI — Complete Graphics Configuration for RG35XX-H (H700)
# Optimized for muOS with 1GB RAM
# ============================================================
# Every option is documented with:
#   - Description         what it does
#   - Parameters          accepted values
#   - Recommend (opt.)    the value we recommend on this hardware
#   - Why (opt.)          short hardware-specific rationale
#
# Per-game overrides go in GameSettings/<GAMEID>.ini using the namespaced
# section format ([Video_Settings], [Video_Enhancements], [Video_Hacks]).
# ============================================================


# ============================================================
# [Hardware] — Graphics hardware
# ============================================================
[Hardware]

# Backend : Graphics API.
# Backend Parameters : OGL (OpenGL ES), Vulkan, D3D11, D3D12, Software, Null.
# Backend Recommend : OGL
# Backend Why : Panfrost (Mesa's Bifrost driver) exposes a mature OpenGL ES 3.1 stack. PanVK is experimental and non-conformant on G31/G51/G52/G76 — do not use it.
Backend = OGL

# Adapter : GPU device index (0 = default).
# Adapter Parameters : Integer (0-N).
Adapter = 0

# VSync : Vertical sync. Caps FPS to the panel refresh rate.
# VSync Parameters : True, False.
# VSync Recommend : False
# VSync Why : Avoids presentation stalls on the Mali driver, reduces input lag.
VSync = False

# BorderlessFullscreen : Use a borderless window instead of exclusive fullscreen.
# BorderlessFullscreen Parameters : True, False.
# BorderlessFullscreen Recommend : False
BorderlessFullscreen = False


# ============================================================
# [Settings] — Core graphics
# ============================================================
[Settings]

# InternalResolution : Render resolution multiplier.
# InternalResolution Parameters : 1 (native), 2 (2x), 3 (3x), ..., 9 (9x).
# InternalResolution Recommend : 1
# InternalResolution Why : The H700's panel is 640x480. Higher IR spends scarce GPU cycles for no visible benefit.
InternalResolution = 1

# EFBScale : Embedded Framebuffer scale factor.
# EFBScale Parameters : 0 (Auto-fractional), 1 (Auto-integral), 2 (1x), 3 (1.5x), 4 (2x).
# EFBScale Recommend : 0
EFBScale = 0

# AspectRatio : Display aspect ratio.
# AspectRatio Parameters : 0 (Auto), 1 (4:3), 2 (16:9), 3 (Stretch).
# AspectRatio Recommend : 0
AspectRatio = 0

# Crop : Crop the image to the active aspect ratio.
# Crop Parameters : True, False.
# Crop Recommend : False
Crop = False

# wideScreenHack : Force widescreen rendering (can break titles).
# wideScreenHack Parameters : True, False.
# wideScreenHack Recommend : False
wideScreenHack = False

# MSAA : Multi-sample anti-aliasing.
# MSAA Parameters : 1 (Off), 2 (2x), 4 (4x), 8 (8x).
# MSAA Recommend : 1
# MSAA Why : Note the off value is 1 (not 0) in Dolphin's numbering. AA is far too expensive on the G31.
MSAA = 1

# SSAA : Super-sample anti-aliasing.
# SSAA Parameters : True, False.
# SSAA Recommend : False
SSAA = False

# ShaderCompilationMode : Shader compilation strategy.
# ShaderCompilationMode Parameters : 0 (Synchronous), 1 (Synchronous Ubershaders), 2 (Async Ubershaders), 3 (Skip Drawing).
# ShaderCompilationMode Recommend : 3
# ShaderCompilationMode Why : Skip Drawing eliminates shader-compilation stutter on Mali entirely. Trade-off: very brief pop-in during extreme transitions.
ShaderCompilationMode = 3

# WaitForShadersBeforeStarting : Wait for all shaders to compile before the game starts.
# WaitForShadersBeforeStarting Parameters : True, False.
# WaitForShadersBeforeStarting Recommend : False
WaitForShadersBeforeStarting = False

# BackendMultithreading : Multi-threaded rendering submission.
# BackendMultithreading Parameters : True, False.
# BackendMultithreading Recommend : True
BackendMultithreading = True

# FastDepthCalc : Fast depth calculation.
# FastDepthCalc Parameters : True, False.
# FastDepthCalc Recommend : True
# FastDepthCalc Why : Despite the name, this is the MORE accurate method on modern Dolphin. Only disable per-game if a title flickers on Panfrost.
FastDepthCalc = True

# EnableGPUTextureDecoding : Offload texture decoding to the GPU.
# EnableGPUTextureDecoding Parameters : True, False.
# EnableGPUTextureDecoding Recommend : True
# EnableGPUTextureDecoding Why : Frees CPU cycles. Worth an A/B test per-game on GPU-bound titles.
EnableGPUTextureDecoding = True

# SafeTextureCacheColorSamples : Texture cache accuracy.
# SafeTextureCacheColorSamples Parameters : 0 (Safe, most accurate), 128 (Fast), 512 (Middle).
# SafeTextureCacheColorSamples Recommend : 0
# SafeTextureCacheColorSamples Why : 0 is the most accurate setting, not a shortcut. Only raise to 512 for a specific title with choppy FMVs.
SafeTextureCacheColorSamples = 0

# EnablePixelLighting : Per-pixel lighting (quality, heavy).

# [Hacks] — Performance hacks
# ============================================================
[Hacks]

# BBoxEnable : Emulate bounding box access.
# BBoxEnable Parameters : True, False.
# BBoxEnable Recommend : False
# BBoxEnable Why : Only a handful of games need it (for hit detection); expensive.
BBoxEnable = False

# DeferEFBCopies Parameters : True, False.
# DeferEFBCopies Recommend : True
DeferEFBCopies = True

# EFBEmulateFormatChanges : Emulate EFB format changes (accuracy).
# EFBEmulateFormatChanges Parameters : True, False.
# EFBEmulateFormatChanges Recommend : False
# EFBEmulateFormatChanges Why : Breaks some Kirby / Zelda titles that rely on format changes for specific effects.
EFBEmulateFormatChanges = False

# EFBScaledCopy : Scale EFB copies.
# EFBScaledCopy Parameters : True, False.
# EFBScaledCopy Recommend : False
# EFBScaledCopy Why : Consistent with native resolution; no wasted GPU work.
EFBScaledCopy = False

# EFBToTextureEnable : Keep EFB copies as GPU textures.
# EFBToTextureEnable Parameters : True, False.
# EFBToTextureEnable Recommend : True
# EFBToTextureEnable Why : Eliminates the CPU readback path.
EFBToTextureEnable = True

# XFBToTextureEnable : Keep XFB copies as GPU textures.
# XFBToTextureEnable Parameters : True, False.
# XFBToTextureEnable Recommend : True
XFBToTextureEnable = True

# ImmediateXFBEnable : Copy XFB immediately.
# ImmediateXFBEnable Parameters : True, False.
# ImmediateXFBEnable Recommend : True
ImmediateXFBEnable = True

# VertexRounding : Round vertex coordinates.
# VertexRounding Parameters : True, False.
# VertexRounding Recommend : True
# VertexRounding Why : Prevents vertex wobble; harmless at native resolution.
VertexRounding = True

# SkipDuplicateXFBs : Skip duplicate XFB copies.
# SkipDuplicateXFBs Parameters : True, False.
# SkipDuplicateXFBs Recommend : True
SkipDuplicateXFBs = True

# EFBAccessEnable : Allow the CPU to read from the EFB (accuracy, heavy).
# EFBAccessEnable Parameters : True, False.
# EFBAccessEnable Recommend : False globally; True per-game if a title relies on EFB reads for gameplay.
EFBAccessEnable = False

# ForceProgressive : Force progressive-scan output.
# ForceProgressive Parameters : True, False.
# ForceProgressive Recommend : False
ForceProgressive = False

# VISkip : Skip some vertex processing.
# VISkip Parameters : True, False.
# VISkip Recommend : False in the baseline.
# VISkip Why : Never enable on Wind Waker or Twilight Princess — they stop rendering entirely with it on. Safe per-game for some CPU-bound titles.

# FastTextureSampling : Use a faster texture sampling path.
